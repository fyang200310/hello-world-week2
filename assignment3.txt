# Ask the user for information
name = input("What is your name? ")
restaurant = input("What is your favorite restaurant? ")
food = input("What is your favorite food? ")
money = float(input("How much money do you have? "))

# Check whether the user has enough money
has_enough_money = money >= 20

# Create a story by joining text and the user's answers
story = (
    name + " went to " + restaurant + " to eat "
    + food + ". "
)

# Add an ending based on the amount of money
if has_enough_money:
    story += name + " had enough money to enjoy a meal there!"
else:
    story += name + " saved some money and planned to visit another day."

# Print the finished story
print(story)
